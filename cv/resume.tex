\begin{rSection}{Research Experience}

\begin{rSubsection}{Research Collaborator}{Feb. 2026 -- Present}{\href{https://liyueshen.engin.umich.edu/}{BioMed-AI Lab, ECE/EECS, University of Michigan} (Advisor: Prof. Liyue Shen)}{Remote}
  \item Led \textbf{MedAction} [1] (COLM 2026): built a tree-structured distillation pipeline with two KG-grounded consistency metrics to synthesize \textbf{MedAction-32K} (32,681 multi-turn diagnostic trajectories from 2,896 PMC cases); the fine-tuned 8B model achieves SOTA among open-source models on MedRBench.
  \item Extending it into \textbf{MedAction-RL}, an SFT + RL recipe for active diagnostic agents that order tests, interpret results, and decide when to stop.
\end{rSubsection}

\begin{rSubsection}{Research Intern}{Nov. 2025 -- Present}{\href{https://khhuang.me}{FLAIR Lab, CSE, Texas A\&M University} (Advisor: Prof. Kuan-Hao Huang)}{Remote}
  \item Research on object hallucination in VLMs from a semantic perspective. Built \textbf{NAME-Bench} [5], a 25K-mention benchmark distinguishing asserted from non-assertive object mentions (e.g., negation, hedging, purpose clauses), and showed that current hallucination detectors conflate mentioning an object with asserting its existence. Also studying cross-modal representations in VLMs via sparse autoencoders.
\end{rSubsection}

\begin{rSubsection}{Research Assistant}{Dec. 2024 -- Present}{\href{https://www.femh.org.tw/mainpage_en/Introduction.aspx}{Far Eastern Memorial Hospital} (Advisor: Dr. Fang-Ming Hung)}{New Taipei, Taiwan}
  \item Designed \textbf{MedPlan} [2] (ACL 2025 Industry Track), a SOAP-aligned two-stage RAG system that generates a clinical assessment and then a patient-specific treatment plan, built on the hospital's extensive real-world EHR data in SOAP format; deployed at the hospital and validated by attending physicians on real patient records (90\%+ ICD-10 prediction accuracy).
  \item Led to a Taiwan utility model patent, the National Innovation Award, and MOUs with clinics and HIS vendors; currently integrating the system with HL7 FHIR for interoperability with hospital information systems.
\end{rSubsection}

\begin{rSubsection}{Undergraduate Capstone Research}{Sep. 2025 -- Present}{Departments of MIS and Computer Science, National Chengchi University}{Taipei, Taiwan}
  \item \textbf{MIS capstone} (Advisor: Prof. Fang Yu): Proposed \textbf{WARP} [6], a provable repair framework for Transformer NLP models beyond the final layer, posed as a convex QP with margin, preservation, and certified-robustness guarantees; extended it to generative LLMs as \textbf{FORGE} [7], which reduces bias and toxicity with minimal perplexity cost and stays stable in low-resource settings where fine-tuning fails. Funded by an \textbf{NSTC Undergraduate Research Grant}.
  \item \textbf{CS capstone} (Advisor: Prof. Shi-Sheng Sun): Distilling the structure of LLM chain-of-thought into a 69K-parameter concept-graph model for explainable vehicle trajectory prediction [8]; matches the strongest baseline with 4.6$\times$ fewer parameters, runs $\sim$1000$\times$ faster than the LLM (0.26 ms/sample), and supports concept-level intervention.
\end{rSubsection}
